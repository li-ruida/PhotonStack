cmake_minimum_required(VERSION 3.20)

file(READ "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" PHOTONSTACK_PROJECT_VERSION)
string(STRIP "${PHOTONSTACK_PROJECT_VERSION}" PHOTONSTACK_PROJECT_VERSION)

execute_process(
  COMMAND git rev-parse --short HEAD
  WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
  OUTPUT_VARIABLE PHOTONSTACK_GIT_COMMIT
  OUTPUT_STRIP_TRAILING_WHITESPACE
  ERROR_QUIET
)
if(PHOTONSTACK_GIT_COMMIT STREQUAL "")
  set(PHOTONSTACK_GIT_COMMIT "unknown")
endif()

execute_process(
  COMMAND git status --porcelain
  WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
  OUTPUT_VARIABLE PHOTONSTACK_GIT_STATUS
  OUTPUT_STRIP_TRAILING_WHITESPACE
  ERROR_QUIET
)
if(NOT PHOTONSTACK_GIT_STATUS STREQUAL "")
  string(APPEND PHOTONSTACK_GIT_COMMIT "-dirty")
endif()

string(TIMESTAMP PHOTONSTACK_BUILD_DATE "%Y-%m-%dT%H:%M:%S%z" UTC)

project(
  PhotonStack
  VERSION ${PHOTONSTACK_PROJECT_VERSION}
  DESCRIPTION "Astrophotography processing engine and CLI"
  LANGUAGES CXX
)

if(APPLE)
  enable_language(OBJCXX)
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_OBJCXX_STANDARD 20)
set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
set(CMAKE_OBJCXX_EXTENSIONS OFF)

option(PHOTONSTACK_BUILD_TESTS "Build PhotonStack tests" ON)

add_subdirectory(engine)
add_subdirectory(apps/PhotonStackCLI)

if(PHOTONSTACK_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests/engine)
  add_subdirectory(tests/cli)
endif()
